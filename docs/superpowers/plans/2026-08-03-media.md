# Media Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement a shared local-disk `StorageService`, then the `Media` module on top of it — multipart photo/video upload to an in-progress inspection, streaming download with an inferred `Content-Type`, and ownership/status-gated delete.

**Architecture:** `StorageModule` is a small, dependency-free `@Injectable()` class (no interface/token abstraction — matches how every other service in this codebase, e.g. `PrismaService`/`SitesService`, is a single concrete class, not an interface+impl pair) exporting `StorageService`. `MediaModule` imports it directly (not globally registered — same non-global pattern `AuditLogModule`/`SitesModule` already use). `MediaController` owns two route prefixes (`/inspections/:inspectionId/media` and `/media/:id`), matching the design's endpoint table exactly.

**Tech Stack:** NestJS 11 (`@nestjs/platform-express`'s `FileInterceptor`, already a dependency), Prisma (no schema migration), `@nestjs/swagger`, `class-validator`/`class-transformer` (all already installed) plus one new dev dependency, `@types/multer`. Package manager is **pnpm**.

## Global Constraints

- No automated tests in this pass. Every task substitutes a manual verification step (build + `pnpm run start:dev` + `curl`) for the usual test-first cycle.
- No Prisma schema migration. `Media` already has every field this module needs; `Media.type` is always derived server-side from the uploaded file's mimetype, never client-supplied.
- Global `JwtAuthGuard` + `RolesGuard` guard every route by default; `@Roles('engineer', 'administrator')` narrows by role, ownership/state checks happen inside the service method — same pattern as `InspectionsService`.
- Every mutation (`upload`/`remove`) calls `AuditLogService.log(userId, action, entityType, entityId, details?, ipAddress?)` — exact signature already defined in `src/audit-log/audit-log.service.ts`.
- Types used only in `@CurrentUser()`-decorated controller parameters must be imported with `import type`, not a plain `import` (this project has `isolatedModules` + `emitDecoratorMetadata` enabled; a plain import used only in a decorated parameter position fails the build with TS1272 — every existing controller already does this correctly).
- When a `class-validator` DTO value (an object or array of a decorated class) is written to a Prisma `Json` column, TypeScript rejects it directly against `Prisma.InputJsonValue` (missing index signature) — cast through `as unknown as Prisma.InputJsonValue`. Not needed in this plan (`Media` has no `Json` column), but note it if a future task adds one.
- Swagger/OpenAPI decorators go directly on the new controller/DTO, consistent with every prior slice.
- Package manager is `pnpm`, not `npm`/`yarn`.

`src/storage/storage.service.ts`:

```typescript
import { Injectable } from '@nestjs/common';
import { createReadStream, promises as fs } from 'fs';
import * as path from 'path';

const UPLOADS_ROOT = path.join(process.cwd(), 'uploads');

@Injectable()
export class StorageService {
  async save(buffer: Buffer, namespace: string, filename: string): Promise<string> {
    const dir = path.join(UPLOADS_ROOT, namespace);
    await fs.mkdir(dir, { recursive: true });
    await fs.writeFile(path.join(dir, filename), buffer);
    return `${namespace}/${filename}`;
  }

  read(relativePath: string): NodeJS.ReadableStream {
    return createReadStream(path.join(UPLOADS_ROOT, relativePath));
  }

  async delete(relativePath: string): Promise<void> {
    try {
      await fs.unlink(path.join(UPLOADS_ROOT, relativePath));
    } catch (error) {
      if ((error as NodeJS.ErrnoException).code !== 'ENOENT') {
        throw error;
      }
    }
  }
}
```

- [x] **Step 3: Write `MediaService`**

`src/media/media.service.ts`:

```typescript
import { ConflictException, ForbiddenException, Injectable, NotFoundException } from '@nestjs/common';
import { randomUUID } from 'crypto';
import * as path from 'path';
import { AuditLogService } from '../audit-log/audit-log.service';
import { AuthenticatedUser } from '../auth/types/jwt-payload.type';
import { PrismaService } from '../prisma/prisma.service';
import { StorageService } from '../storage/storage.service';
import { UploadMediaDto } from './dto/upload-media.dto';

const MEDIA_NAMESPACE = 'media';

@Injectable()
export class MediaService {
  constructor(
    private prisma: PrismaService,
    private storageService: StorageService,
    private auditLogService: AuditLogService,
  ) {}

  findAllForInspection(inspectionId: string) {
    return this.prisma.media.findMany({
      where: { inspectionId },
      orderBy: { createdAt: 'desc' },
    });
  }

  async upload(
    inspectionId: string,
    file: Express.Multer.File,
    dto: UploadMediaDto,
    user: AuthenticatedUser,
    ip?: string,
  ) {
    const inspection = await this.prisma.inspection.findUnique({ where: { id: inspectionId } });
    if (!inspection) {
      throw new NotFoundException('Inspection not found');
    }
    this.assertOwnerOrAdmin(inspection.engineerId, user);
    if (inspection.status !== 'in_progress') {
      throw new ConflictException('Media can only be added to an in-progress inspection');
    }

    const type = file.mimetype.startsWith('video/') ? 'video' : 'photo';
    const extension = path.extname(file.originalname);
    const filename = `${randomUUID()}${extension}`;
    const url = await this.storageService.save(file.buffer, MEDIA_NAMESPACE, filename);

    const media = await this.prisma.media.create({
      data: {
        inspectionId,
        type,
        url,
        latitude: dto.latitude,
        longitude: dto.longitude,
        caption: dto.caption,
      },
    });

    await this.auditLogService.log(user.userId, 'media.uploaded', 'Media', media.id, { inspectionId, type }, ip);

    return media;
  }

  async getFileStream(id: string) {
    const media = await this.findMediaOrThrow(id);
    return { stream: this.storageService.read(media.url), extension: path.extname(media.url) };
  }

  async remove(id: string, user: AuthenticatedUser, ip?: string) {
    const media = await this.findMediaOrThrow(id);
    const inspection = await this.prisma.inspection.findUniqueOrThrow({ where: { id: media.inspectionId } });
    this.assertOwnerOrAdmin(inspection.engineerId, user);

    await this.storageService.delete(media.url);
    await this.prisma.media.delete({ where: { id } });

    await this.auditLogService.log(
      user.userId,
      'media.deleted',
      'Media',
      id,
      { inspectionId: media.inspectionId },
      ip,
    );
  }

  private assertOwnerOrAdmin(engineerId: string, user: AuthenticatedUser) {
    if (engineerId !== user.userId && !user.roles.includes('administrator')) {
      throw new ForbiddenException('You do not own this inspection');
    }
  }

  private async findMediaOrThrow(id: string) {
    const media = await this.prisma.media.findUnique({ where: { id } });
    if (!media) {
      throw new NotFoundException('Media not found');
    }
    return media;
  }
}
```

- [x] **Step 4: Write `MediaController`**

`src/media/media.controller.ts`:

```typescript
import {
  Body,
  Controller,
  Delete,
  Get,
  Ip,
  Param,
  Post,
  Res,
  UnsupportedMediaTypeException,
  UploadedFile,
  UseInterceptors,
} from '@nestjs/common';
import { FileInterceptor } from '@nestjs/platform-express';
import { ApiBearerAuth, ApiBody, ApiConsumes, ApiOperation, ApiTags } from '@nestjs/swagger';
import type { Response } from 'express';
import type { AuthenticatedUser } from '../auth/types/jwt-payload.type';
import { CurrentUser } from '../common/decorators/current-user.decorator';
import { Roles } from '../common/decorators/roles.decorator';
import { UploadMediaDto } from './dto/upload-media.dto';
import { MediaService } from './media.service';

const CONTENT_TYPES: Record<string, string> = {
  '.jpg': 'image/jpeg',
  '.jpeg': 'image/jpeg',
  '.png': 'image/png',
  '.gif': 'image/gif',
  '.webp': 'image/webp',
  '.mp4': 'video/mp4',
  '.mov': 'video/quicktime',
  '.webm': 'video/webm',
};

@ApiTags('Media')
@ApiBearerAuth()
@Controller()
export class MediaController {
  constructor(private mediaService: MediaService) {}

  @ApiOperation({ summary: 'List all media for an inspection' })
  @Get('inspections/:inspectionId/media')
  findAllForInspection(@Param('inspectionId') inspectionId: string) {
    return this.mediaService.findAllForInspection(inspectionId);
  }

  @ApiOperation({ summary: 'Upload a photo or video to an in-progress inspection (owning engineer or administrator)' })
  @ApiConsumes('multipart/form-data')
  @ApiBody({
    schema: {
      type: 'object',
      properties: {
        file: { type: 'string', format: 'binary' },
        latitude: { type: 'number' },
        longitude: { type: 'number' },
        caption: { type: 'string' },
      },
      required: ['file', 'latitude', 'longitude'],
    },
  })
  @Roles('engineer', 'administrator')
  @UseInterceptors(
    FileInterceptor('file', {
      limits: { fileSize: 25 * 1024 * 1024 },
      fileFilter: (_req, file, callback) => {
        if (!/^(image|video)\//.test(file.mimetype)) {
          callback(new UnsupportedMediaTypeException('Only image or video uploads are allowed'), false);
          return;
        }
        callback(null, true);
      },
    }),
  )
  @Post('inspections/:inspectionId/media')
  upload(
    @Param('inspectionId') inspectionId: string,
    @UploadedFile() file: Express.Multer.File,
    @Body() dto: UploadMediaDto,
    @CurrentUser() user: AuthenticatedUser,
    @Ip() ip: string,
  ) {
    return this.mediaService.upload(inspectionId, file, dto, user, ip);
  }

  @ApiOperation({ summary: 'Stream a media file' })
  @Get('media/:id/file')
  async getFile(@Param('id') id: string, @Res() res: Response) {
    const { stream, extension } = await this.mediaService.getFileStream(id);
    res.set('Content-Type', CONTENT_TYPES[extension] ?? 'application/octet-stream');
    stream.on('error', () => {
      if (!res.headersSent) {
        res.status(404).json({ statusCode: 404, message: 'Media file not found', error: 'Not Found' });
      }
    });
    stream.pipe(res);
  }

  @ApiOperation({ summary: 'Delete a media item (owning engineer or administrator)' })
  @Roles('engineer', 'administrator')
  @Delete('media/:id')
  remove(@Param('id') id: string, @CurrentUser() user: AuthenticatedUser, @Ip() ip: string) {
    return this.mediaService.remove(id, user, ip);
  }
}
```
